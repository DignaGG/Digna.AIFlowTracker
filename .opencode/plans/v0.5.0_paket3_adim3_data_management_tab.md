Import (button-tetiklemeli; mod yanlisligi onlenir):
```ts
const handleImportClick = useCallback(async () => {
  if (!selectedFile) return
  setImportStatus({ kind: 'loading' })
  const reader = new FileReader()
  reader.onload = async () => {
    try {
      const text = String(reader.result ?? '')
      const result = await importDataFromJson(text, importMode)
      if (result.success) {
        setImportStatus({ kind: 'success', count: result.importedCount })
        onDataImported()
      } else {
        setImportStatus({ kind: 'error', message: result.errorMessage ?? '' })
      }
    } catch (err) {
      setImportStatus({ kind: 'error', message: err instanceof Error ? err.message : t.settings.data.importError })
    } finally {
      setSelectedFile(null)
      if (fileInputRef.current) fileInputRef.current.value = ''
    }
  }
  reader.onerror = () => {
    setImportStatus({ kind: 'error', message: t.settings.data.importError })
    setSelectedFile(null)
    if (fileInputRef.current) fileInputRef.current.value = ''
  }
  reader.readAsText(selectedFile)
}, [selectedFile, importMode, onDataImported, t])
```
